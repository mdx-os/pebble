# Talking to the agent: hold to talk, let go to send, and the Talk shortcut.

# The request no longer asks for on-device recognition, so speech could go
# to a server.
test: speechIsHeardOnTheDeviceOnlyAndNeverKept
file: Packages/PebbleKit/Sources/PebbleVoice/DeviceSpeech.swift
perl: s~\n        request\.requiresOnDeviceRecognition = true~~

# The same, turned off rather than left out.
test: speechIsHeardOnTheDeviceOnlyAndNeverKept
file: Packages/PebbleKit/Sources/PebbleVoice/DeviceSpeech.swift
perl: s~request\.requiresOnDeviceRecognition = true~request.requiresOnDeviceRecognition = false~

# The capture asks for a listen that may leave the device.
test: noServerFallbackIsEverRequested
file: Packages/PebbleKit/Sources/PebbleCore/VoiceCapture.swift
perl: s~ListenRequest\(onDeviceOnly: true\)~ListenRequest(onDeviceOnly: false)~

# A cancel sends what was heard.
test: cancelSendsNothing
file: Packages/PebbleKit/Sources/PebbleCore/VoiceCapture.swift
perl: s~        target\.write\(target\.before\)\n        stop\(\.cancelled\)~        target.send(heard)\n        target.write(target.before)\n        stop(.cancelled)~
